Что такое PL/SQL?

Процедурное расширение SQL от Oracle — объединяет мощь баз данных с логикой программирования

Зачем нужен PL/SQL, если есть SQL?

PL/SQL умеет

Процедурную логику прямо внутри базы данных, сохранённые процедуры и триггеры

Ключевые характеристики

ХарактеристикаОписание
ПарадигмаПроцедурный язык, ориентированный на обработку данных
КомпиляцияКомпилируемый язык — код проверяется до выполнения
Блочная структураКод организован в блоки: DECLARE → BEGIN → EXCEPTION → END

Аналоги в других СУБД: PostgreSQL → PL/pgSQL, Microsoft SQL Server → T-SQL, MySQL → собственный диалект. Все работают схоже, но синтаксис отличается.

Структура PL/SQL блока

Любой PL/SQL код организован в блоки с фиксированной структурой

Анатомия блока

[DECLARE
v_переменная тип; -- объявление переменных
CURSOR имя IS SELECT ...; -- курсоры
e_исключение EXCEPTION; ] -- пользовательские исключения

BEGIN
-- обязательная секция
SQL_инструкции;
PL/SQL_инструкции;

[EXCEPTION
WHEN имя_исключения THEN
обработка_ошибки; ]

END;
/ -- запуск блока в SQL*Plus / SQL Developer

DECLARE необязательно

Объявление переменных, констант, курсоров, исключений. Всё, что используется в блоке

BEGIN обязательно

Основная логика: SQL-запросы, операторы, вызовы подпрограмм

EXCEPTION необязательно

Обработка ошибок, которые возникли в секции BEGIN

END; обязательно

Завершает блок. Символ / на отдельной строке выполняет блок

Минимальный рабочий блок

BEGIN
DBMS_OUTPUT.PUT_LINE('Привет, PL/SQL!');
END;
/

Переменные и константы

Хранение и обработка данных внутри PL/SQL блока

Синтаксис объявления

имя [CONSTANT] тип [NOT NULL] [{:= | DEFAULT} значение];

Примеры

DECLARE
v_age NUMBER(3); -- просто объявление
v_name VARCHAR2(30) := 'Анна'; -- с начальным значением
v_counter NUMBER(4) NOT NULL := 0; -- не может быть NULL
v_sum NUMBER(10) DEFAULT 0; -- через DEFAULT
c_pi CONSTANT NUMBER := 3.1415; -- константа
BEGIN
v_age := 25; -- присвоение в BEGIN
DBMS_OUTPUT.PUT_LINE(v_name || ', возраст: ' || v_age);
END;
/
Правила имён переменных: начинается с буквы, содержит буквы/цифры/_/$/#, не более 30 символов, не совпадает с ключевыми словами SQL/PL/SQL, не содержит пробелов.
Принято использовать префиксы: v_ для переменных, c_ для констант.

Специальные символы в строках

-- Экранирование апострофа — удвоить одинарную кавычку
v_str := 'I''m learning';

-- Нотация q' — указываем свой разделитель
v_str := q'!I'm listening!';
v_str := q'[I'm bored]';

Скалярные типы

КатегорияТипыПример
ЧисловыеNUMBER, INTEGER, FLOAT, BINARY_INTEGERv_x NUMBER(10,2)
СимвольныеVARCHAR2, CHAR, NVARCHAR2v_name VARCHAR2(50)
ЛогическийBOOLEANv_flag BOOLEAN := TRUE

Важно: тип BOOLEAN существует только в PL/SQL — в обычных SQL-запросах его нет. Значения: TRUE, FALSE, NULL.

Преобразование типов

-- Явное преобразование (надёжнее)
v_str := TO_CHAR(v_num); -- число → строка
v_num := TO_NUMBER('42'); -- строка → число
v_dt := TO_DATE('01.01.2024','DD.MM.YYYY');

-- Неявное (автоматически, но может вызвать ошибки!)
v_num := '100'; -- Oracle сам преобразует строку в число

%TYPE — тип из столбца таблицы

Переменная получает тот же тип, что и столбец в базе данных. Это защищает от несоответствия типов.

DECLARE
-- v_name получает тип колонки employees.first_name
v_name employees.first_name%TYPE;
v_sal employees.salary%TYPE;
BEGIN
SELECT first_name, salary
INTO v_name, v_sal
FROM employees
WHERE employee_id = 100;
END;
/

%ROWTYPE — тип строки таблицы

Переменная-запись, структура которой совпадает со строкой таблицы.

DECLARE
vr_emp employees%ROWTYPE; -- все поля строки employees
BEGIN
SELECT * INTO vr_emp
FROM employees
WHERE employee_id = 100;

DBMS_OUTPUT.PUT_LINE(vr_emp.first_name || ' ' || vr_emp.last_name);
END;
/

RECORD — пользовательская запись

DECLARE
TYPE r_address IS RECORD (

Область видимости переменных

Переменная видна только в том блоке, где она объявлена, и во вложенных блоках.

DECLARE
v_outer NUMBER := 10; -- видна везде
BEGIN
DECLARE
v_inner NUMBER := 20; -- видна только здесь
BEGIN
DBMS_OUTPUT.PUT_LINE(v_outer + v_inner); -- OK: 30
END;
-- v_inner здесь уже НЕ видна
DBMS_OUTPUT.PUT_LINE(v_outer); -- OK: 10
END;
/

SQL внутри PL/SQL

Как встраивать запросы и DML-операции в блоки PL/SQL

SELECT INTO — выборка в переменную

Три жёстких правила SELECT INTO:
1. Запрос должен вернуть ровно одну строку (иначе: TOO_MANY_ROWS или NO_DATA_FOUND)
2. Предложение INTO обязательно
3. Имена столбцов БД имеют приоритет над именами переменных
DECLARE
v_name employees.first_name%TYPE;
v_salary employees.salary%TYPE;
BEGIN
SELECT first_name, salary
INTO v_name, v_salary
FROM employees
WHERE employee_id = 200;

DBMS_OUTPUT.PUT_LINE('Имя: ' || v_name || ', Зарплата: ' || v_salary);
END;
/

DML операции (INSERT, UPDATE, DELETE)

DECLARE
v_new_id departments.department_id%TYPE;
BEGIN
-- INSERT с RETURNING — возвращает значение вставленной строки
INSERT INTO departments (department_id, department_name)
VALUES (300, 'Analytics')
RETURNING department_id INTO v_new_id;

-- UPDATE как обычно
UPDATE employees
SET salary = salary * 1.1
WHERE department_id = v_new_id;

-- Фиксируем транзакцию
COMMIT;
END;
/

Управляющие конструкции

Условия и циклы — то, ради чего PL/SQL и был создан

IF / ELSIF
CASE
Циклы

Три формы IF

IF условие THEN
инструкции;
END IF;

Выполняет инструкции, только если условие истинно (TRUE). Если FALSE или NULL — пропускает.

IF условие THEN
инструкции_1;
ELSE
инструкции_2;
END IF;

ELSE выполняется при любом результате, отличном от TRUE (в т.ч. при NULL).

IF условие_1 THEN
инструкции_1;
ELSIF условие_2 THEN
инструкции_2;
ELSIF условие_3 THEN
инструкции_3;
ELSE
инструкции_n;
END IF;

Пишется ELSIF, а не ELSEIF! Условия проверяются по очереди — выполняется первое подходящее.

Практический пример

DECLARE
v_score NUMBER := 75;
v_grade VARCHAR2(1);
BEGIN
IF v_score >= 90 THEN v_grade := 'A';
ELSIF v_score >= 80 THEN v_grade := 'B';
ELSIF v_score >= 70 THEN v_grade := 'C';
ELSE v_grade := 'F';
END IF;
DBMS_OUTPUT.PUT_LINE('Оценка: ' || v_grade);
END;
/

CASE — две формы

-- Форма 1: простой CASE (проверяет равенство)
CASE v_day
WHEN 1 THEN DBMS_OUTPUT.PUT_LINE('Понедельник');
WHEN 2 THEN DBMS_OUTPUT.PUT_LINE('Вторник');
WHEN 7 THEN DBMS_OUTPUT.PUT_LINE('Воскресенье');
ELSE DBMS_OUTPUT.PUT_LINE('Другой день');
END CASE;

-- Форма 2: поисковый CASE (проверяет условия)
CASE
WHEN v_salary < 3000 THEN v_level := 'Junior';
WHEN v_salary < 7000 THEN v_level := 'Middle';
ELSE v_level := 'Senior';
END CASE;
Если ни одна ветка CASE не подошла и нет ELSE — возникает исключение CASE_NOT_FOUND.

Три вида циклов

ЦиклКогда использовать
LOOP ... END LOOPТело должно выполниться хотя бы один раз
WHILE ... LOOPУсловие проверяется до каждой итерации
FOR i IN 1..n LOOPЧисло итераций известно заранее
-- LOOP с EXIT
LOOP
v_i := v_i + 1;
EXIT WHEN v_i > 10; -- выход когда условие TRUE
END LOOP;

-- WHILE
WHILE v_i <= 10 LOOP
v_sum := v_sum + v_i;
v_i := v_i + 1;
END LOOP;

-- FOR (счётчик i создаётся автоматически, менять нельзя!)
FOR i IN 1..10 LOOP
DBMS_OUTPUT.PUT_LINE(i);
END LOOP;

-- FOR в обратном порядке
FOR i IN REVERSE 1..10 LOOP
DBMS_OUTPUT.PUT_LINE(i); -- 10, 9, 8, ... 1
END LOOP;

CONTINUE — пропуск итерации

FOR i IN 1..10 LOOP
-- Пропускаем чётные числа
CONTINUE WHEN MOD(i, 2) = 0;
DBMS_OUTPUT.PUT_LINE('Нечётное: ' || i);
END LOOP;

Курсоры

Как обрабатывать многострочные результаты запросов

Курсор — указатель на приватную область памяти Oracle, которая хранит результат выполнения SELECT или DML. Позволяет обходить результаты строка за строкой.

Неявные курсоры

Создаются автоматически для всех DML и SELECT INTO. Управляются Oracle

Явные курсоры

Создаются программистом для SELECT, возвращающих несколько строк. Нужно самому управлять: OPEN → FETCH → CLOSE

Ручное управление
Cursor FOR Loop
Атрибуты

Полный цикл: объявление → открытие → выборка → закрытие

DECLARE
-- 1. Объявление курсора
CURSOR c_emp (p_dept_id NUMBER) IS
SELECT * FROM employees
WHERE department_id = p_dept_id;

vr_emp employees%ROWTYPE;
BEGIN
-- 2. Открытие курсора с параметром
OPEN c_emp(50);

LOOP
-- 3. Выборка строки
FETCH c_emp INTO vr_emp;
-- Выход когда строк больше нет
EXIT WHEN c_emp%NOTFOUND OR c_emp%NOTFOUND IS NULL;

DBMS_OUTPUT.PUT_LINE(c_emp%ROWCOUNT || '. ' || vr_emp.last_name);
END LOOP;

DBMS_OUTPUT.PUT_LINE('Всего: ' || c_emp%ROWCOUNT);

-- 4. Закрытие курсора
CLOSE c_emp;
END;
/

Cursor FOR Loop — самый краткий способ

Oracle автоматически открывает, выбирает и закрывает курсор. Рекомендуется как основной подход.

-- Вариант 1: с именованным курсором
DECLARE
CURSOR c_emp (p_id NUMBER) IS
SELECT * FROM employees WHERE department_id = p_id;
BEGIN
FOR vr_emp IN c_emp(50) LOOP
DBMS_OUTPUT.PUT_LINE(vr_emp.last_name);
END LOOP;
END;
/

-- Вариант 2: с подзапросом (ещё короче)
BEGIN
FOR vr_emp IN (SELECT * FROM employees WHERE department_id = 50) LOOP
DBMS_OUTPUT.PUT_LINE(vr_emp.last_name);
END LOOP;
END;
/
Cursor FOR Loop — предпочтительный способ в большинстве случаев: код короче, нет риска забыть CLOSE.

Атрибуты явного курсора

АтрибутВозвращаетДо первого FETCH
%ISOPENTRUE если курсор открытFALSE
%FOUNDTRUE если последний FETCH вернул строкуNULL
%NOTFOUNDTRUE если строк больше нетNULL
%ROWCOUNTКоличество выбранных строк0
Важно для EXIT WHEN: до первого FETCH атрибут %NOTFOUND равен NULL, а не FALSE. Поэтому правильно писать:
EXIT WHEN c_emp%NOTFOUND OR c_emp%NOTFOUND IS NULL;

Обработка исключений

Как перехватывать и обрабатывать ошибки в PL/SQL

Структура обработки

BEGIN
рискованные_инструкции;
EXCEPTION
WHEN TOO_MANY_ROWS THEN
DBMS_OUTPUT.PUT_LINE('Слишком много строк!');
WHEN NO_DATA_FOUND THEN
DBMS_OUTPUT.PUT_LINE('Данные не найдены');
WHEN OTHERS THEN -- перехватывает всё остальное (последним!)
DBMS_OUTPUT.PUT_LINE('Ошибка: ' || SQLERRM);
END;
/

Стандартные исключения Oracle

NO_DATA_FOUNDSELECT INTO не вернул ни одной строки
TOO_MANY_ROWSSELECT INTO вернул более одной строки
ZERO_DIVIDEДеление на ноль
DUP_VAL_ON_INDEXНарушение уникальности (уникальный индекс)
VALUE_ERRORАрифметическая ошибка, ошибка преобразования или переполнение
INVALID_CURSORПопытка закрыть незакрытый или несуществующий курсор
CASE_NOT_FOUNDНет подходящей ветки WHEN в CASE без ELSE
CURSOR_ALREADY_OPENПопытка открыть уже открытый курсор

Пользовательские исключения

DECLARE
-- 1. Объявляем
e_invalid_id EXCEPTION;
v_id NUMBER := 0;
BEGIN
-- 2. Генерируем (RAISE)
IF v_id = 0 THEN
RAISE e_invalid_id;
END IF;
EXCEPTION
-- 3. Перехватываем
WHEN e_invalid_id THEN
DBMS_OUTPUT.PUT_LINE('Ошибка: ID не может быть нулевым');
END;
/

RAISE_APPLICATION_ERROR

Позволяет генерировать ошибку с собственным кодом и сообщением из хранимых подпрограмм.

-- Код ошибки: от -20000 до -20999
IF v_id = 0 THEN
RAISE_APPLICATION_ERROR(-20001, 'ID не может быть нулевым');
END IF;

Непредопределённые исключения Oracle

DECLARE
-- Связываем имя с кодом ошибки Oracle
e_insert EXCEPTION;
PRAGMA EXCEPTION_INIT(e_insert, -01400); -- ORA-01400: NULL в NOT NULL
BEGIN
INSERT INTO departments (department_id, department_name)
VALUES (400, NULL);
EXCEPTION
WHEN e_insert THEN
DBMS_OUTPUT.PUT_LINE('Ошибка вставки: NULL в обязательном поле');
END;
/
